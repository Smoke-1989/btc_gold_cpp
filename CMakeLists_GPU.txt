# ============================================================================
# BTC GOLD C++ - GPU/CUDA ACCELERATION BUILD CONFIGURATION
# ============================================================================
# 
# CLASSIFICATION: ENTERPRISE PRODUCTION
# SECURITY LEVEL: Governmental Grade (Level 5)
#
# This CMakeLists configuration enables GPU acceleration via NVIDIA CUDA.
# Maintains full compatibility with CPU-only builds via conditional compilation.
#
# Build Instructions:
#   mkdir build_gpu && cd build_gpu
#   cmake -DENABLE_GPU=ON -DCUDA_TOOLKIT_ROOT_DIR=/usr/local/cuda ..
#   cmake --build . --config Release
#
# ============================================================================

cmake_minimum_required(VERSION 3.18 FATAL_ERROR)

project(
    btc_gold_cpp_gpu
    VERSION 6.0.0
    DESCRIPTION "Enterprise-grade Bitcoin key recovery with GPU acceleration"
    LANGUAGES CXX CUDA
)

# ============================================================================
# PROJECT CONFIGURATION
# ============================================================================

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)

# GPU Enable Option
option(ENABLE_GPU "Enable GPU acceleration" ON)
option(ENABLE_DEBUG_GPU "Enable GPU debug logging" OFF)
option(ENABLE_CUDA_SANITIZER "Enable CUDA memory sanitizer" OFF)

# Compute Capability (defaults to ampere - RTX 30 series)
set(CUDA_COMPUTE_CAPABILITY "80" CACHE STRING "CUDA compute capability (70=Volta, 75=Turing, 80=Ampere, 90=Hopper)")

# ============================================================================
# FIND CUDA
# ============================================================================

find_package(CUDA 11.0 REQUIRED)

if(NOT CUDA_FOUND)
    message(FATAL_ERROR "CUDA 11.0+ is required for GPU acceleration")
endif()

message(STATUS "CUDA Version: ${CUDA_VERSION}")
message(STATUS "CUDA Toolkit Root: ${CUDA_TOOLKIT_ROOT_DIR}")

# ============================================================================
# COMPILER FLAGS
# ============================================================================

# C++ Compiler Flags
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -march=native -DNDEBUG")
set(CMAKE_CXX_FLAGS_DEBUG "-g -O0 -Wall -Wextra")

# CUDA Compiler Flags
set(CUDA_NVCC_FLAGS "-Xptxas=-O3 -Xcompiler -O3")
set(CUDA_NVCC_FLAGS "${CUDA_NVCC_FLAGS} -gencode=arch=compute_${CUDA_COMPUTE_CAPABILITY},code=sm_${CUDA_COMPUTE_CAPABILITY}")
set(CUDA_NVCC_FLAGS "${CUDA_NVCC_FLAGS} --expt-relaxed-constexpr")
set(CUDA_NVCC_FLAGS "${CUDA_NVCC_FLAGS} -maxrregcount=128")  # Limit registers

if(ENABLE_DEBUG_GPU)
    set(CUDA_NVCC_FLAGS "${CUDA_NVCC_FLAGS} -g -G")  # Device debugging
    add_definitions(-DENABLE_GPU_DEBUG_LOGGING)
endif()

if(ENABLE_CUDA_SANITIZER)
    set(CUDA_NVCC_FLAGS "${CUDA_NVCC_FLAGS} -Xcompiler -fsanitize=address")
endif()

# ============================================================================
# SOURCE FILES
# ============================================================================

# CPU Source Files (existing)
set(CPU_SOURCES
    src/main_v5.cpp
    src/worker_v5_complete.cpp
    src/worker_engine.cpp
    src/interactive_menu.cpp
    src/database.cpp
    src/secp256k1_wrapper.cpp
    src/hash160.cpp
    src/logger.cpp
    src/benchmark.cpp
)

# GPU Source Files (new)
set(GPU_SOURCES
    src/gpu_engine.cpp
    src/cuda/gpu_kernels.cu
)

if(ENABLE_GPU)
    list(APPEND SOURCES ${GPU_SOURCES})
    add_definitions(-DENABLE_GPU_ACCELERATION)
    message(STATUS "GPU Acceleration: ENABLED")
else()
    message(STATUS "GPU Acceleration: DISABLED")
endif()

list(APPEND SOURCES ${CPU_SOURCES})

# ============================================================================
# INCLUDE DIRECTORIES
# ============================================================================

include_directories(
    ${PROJECT_SOURCE_DIR}/include
    ${CUDA_INCLUDE_DIRS}
)

# ============================================================================
# LIBRARY CONFIGURATION
# ============================================================================

# CUDA Runtime
if(ENABLE_GPU)
    set(CUDA_LIBRARIES ${CUDA_LIBRARIES} -lcuda -lcudart)
endif()

# ============================================================================
# EXECUTABLE
# ============================================================================

if(ENABLE_GPU)
    # Use CUDA_ADD_EXECUTABLE for mixed CUDA/C++ compilation
    cuda_add_executable(btc_gold_cuda
        ${SOURCES}
    )
    
    target_link_libraries(btc_gold_cuda
        ${CUDA_LIBRARIES}
        ${CUDA_CUBLAS_LIBRARIES}
        pthread
    )
else()
    add_executable(btc_gold
        ${SOURCES}
    )
    
    target_link_libraries(btc_gold
        pthread
    )
endif()

# ============================================================================
# COMPILER PROPERTIES
# ============================================================================

if(ENABLE_GPU)
    set_target_properties(btc_gold_cuda PROPERTIES
        CXX_STANDARD 17
        CXX_STANDARD_REQUIRED ON
        CUDA_STANDARD 17
        CUDA_SEPARABLE_COMPILATION ON
        CUDA_ARCHITECTURES "${CUDA_COMPUTE_CAPABILITY}"
    )
else()
    set_target_properties(btc_gold PROPERTIES
        CXX_STANDARD 17
        CXX_STANDARD_REQUIRED ON
    )
endif()

# ============================================================================
# OPTIMIZATION FLAGS
# ============================================================================

if(CMAKE_BUILD_TYPE MATCHES Release)
    if(ENABLE_GPU)
        set_target_properties(btc_gold_cuda PROPERTIES
            COMPILE_FLAGS "-O3 -march=native"
        )
    else()
        set_target_properties(btc_gold PROPERTIES
            COMPILE_FLAGS "-O3 -march=native"
        )
    endif()
endif()

# ============================================================================
# OUTPUT CONFIGURATION
# ============================================================================

if(ENABLE_GPU)
    set_target_properties(btc_gold_cuda PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
    )
    
    # Install GPU binary
    install(TARGETS btc_gold_cuda
        RUNTIME DESTINATION bin
    )
else()
    set_target_properties(btc_gold PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
    )
    
    # Install CPU binary
    install(TARGETS btc_gold
        RUNTIME DESTINATION bin
    )
endif()

# ============================================================================
# BUILD INFORMATION
# ============================================================================

message(STATUS "")
message(STATUS "=== BTC GOLD GPU BUILD CONFIGURATION ===")
message(STATUS "GPU Acceleration: ${ENABLE_GPU}")
if(ENABLE_GPU)
    message(STATUS "CUDA Version: ${CUDA_VERSION}")
    message(STATUS "CUDA Compute Capability: ${CUDA_COMPUTE_CAPABILITY}")
    message(STATUS "Debug GPU Logging: ${ENABLE_DEBUG_GPU}")
    message(STATUS "CUDA Sanitizer: ${ENABLE_CUDA_SANITIZER}")
endif()
message(STATUS "Build Type: ${CMAKE_BUILD_TYPE}")
message(STATUS "C++ Standard: ${CMAKE_CXX_STANDARD}")
if(ENABLE_GPU)
    message(STATUS "CUDA Standard: ${CMAKE_CUDA_STANDARD}")
endif()
message(STATUS "")
message(STATUS "To build:")
if(ENABLE_GPU)
    message(STATUS "  cmake --build . --config ${CMAKE_BUILD_TYPE}")
    message(STATUS "  Output: ${CMAKE_BINARY_DIR}/bin/btc_gold_cuda")
else()
    message(STATUS "  cmake --build . --config ${CMAKE_BUILD_TYPE}")
    message(STATUS "  Output: ${CMAKE_BINARY_DIR}/bin/btc_gold")
endif()
message(STATUS "")
